Build Your Own SPECTER Node

A wireless field sensor you can build yourself. The SPECTER Node pairs a capacitive proximity antenna with a 3-axis magnetometer on an ESP32, then streams live EMF and proximity telemetry straight into SPECTER over Wi-Fi — appearing in the instrument rail and the 3D field mesh alongside your depth and thermal sensors. Total build cost is roughly $15–25 in parts.

What it looks like in SPECTER

The SPECTER Node view is designed for your phone — you carry it in the field. This is a live simulation: a real-time 3D field radar that shows not just how strong the magnetic disturbance is, but which direction it's coming from. It rotates so you can read the field in 3D, and locks a bearing on the source. Rotate your phone (or resize) and it flips to landscape automatically.

How it works

The node is a small battery- or USB-powered pod. An ESP32 microcontroller reads two sensors a hundred times a second: a capacitive telescopic antenna (its reading drops as a person or object approaches — a proximity/field-disturbance channel) and a QMC5883L magnetometer (a 3-axis reading of the local magnetic field). An onboard LED ring and buzzer give you a standalone alert, while the ESP32 joins your Wi-Fi and connects to the SPECTER desktop app.

SPECTER runs a small server that the node discovers automatically — the app broadcasts its address, the node finds it and connects, and telemetry starts flowing at 10 Hz. From there the node behaves like any other SPECTER sensor: its readings feed the Anomaly Index, threshold crossings fire timestamped detection events into your evidence archive, and the live field mesh renders in the instrument rail.

No soldering required for a basic build. Using breakout boards and jumper wires you can assemble a working node on a breadboard. Soldering only comes in when you want a permanent, rugged pod.

Any ESP32 dev board with the standard pinout works. The QMC5883L is the common blue "GY-273 / HMC5883L-style" module — confirm it's the QMC5883L variant (I2C address 0x0D).

Wiring

Connect each component to the ESP32 as below. All sensors run on 3.3V. The QMC5883L uses the I2C bus (SDA/SCL); the LED ring, buzzer, and antenna each take one GPIO pin.

Component pinESP32 pin
Telescopic antenna wireGPIO4 (T0, touch)
QMC5883L · SDAGPIO21
QMC5883L · SCLGPIO22
QMC5883L · VCC / GND3.3V / GND
LED ring · DINGPIO13
LED ring · VCC / GND3.3V (or 5V) / GND
Buzzer · +GPIO12
Buzzer · −GND
The antenna is just a length of wire or a telescopic rod connected to GPIO4. The ESP32's built-in capacitive touch sensing on that pin reads the change in capacitance as something approaches — no extra components needed.

The code

This is the complete firmware in a single file — the on-device dashboard (live telemetry, the 3D field-vector graph, Wi-Fi setup and settings) is built right into it. No separate data upload, no extra plugins. Just paste it into the Arduino IDE, install the three libraries below, select your ESP32 board, and hit Upload.

After it uploads, the node hosts a Wi-Fi network called SPECTER-NODE-01. Join it from your phone, open 192.168.4.1 in a browser, and you’ll land on the live telemetry page. Tap Settings to connect the node to your home Wi-Fi and tune it.

// ===== LED CONFIG =====
// Generate this block with the configurator at
//   specter-imaging.com/help/build-node/#leds  (Configure your LEDs)
// then paste over these lines. Defaults suit a 12-pixel WS2812B ring.
#define LED_CHIPSET   WS2812B
#define LED_COLOR_ORD GRB
#define LED_DATA_PIN  13
// #define LED_CLK_PIN   14   // uncomment for clocked chips (APA102 / SK9822)
#define NUM_LEDS      12
#define LED_BRIGHT    100

// Positional map — which LED index points which way (-1 = unset).
// Face the antenna FORWARD; SPECTER lights the side toward a source.
#define LED_FRONT     0
#define LED_RIGHT     3
#define LED_BACK      6
#define LED_LEFT      9

#define APP_WS_PORT   47846

// ---- hardware globals ----
CRGB leds[NUM_LEDS];
Preferences prefs;
volatile int16_t magX = 0, magY = 0, magZ = 0;
volatile int     proxVal = 0;
volatile int     magDelta = 0;
volatile bool    eventPending = false;
volatile int     evtValue = 0, evtThreshold = 0;
char             evtChannel[8] = "mag", evtSeverity[8] = "low", evtKind[16] = "spike";
volatile int16_t baseMagX = 0, baseMagY = 0, baseMagZ = 0;
volatile int     baseProx = 0;
volatile bool    recalibrate = true;
volatile int     threshProx = 2000;   // capacitance DROPS as an object nears
volatile int     threshMag  = 1500;   // vector delta (LSB)
volatile int     telemetryHz = TELEMETRY_HZ_DEFAULT;

#define QMC5883L_ADDR 0x0D

void initSensors() {
    Wire.begin(I2C_SDA, I2C_SCL);
    Wire.beginTransmission(QMC5883L_ADDR); Wire.write(0x09); Wire.write(0x1D); Wire.endTransmission();
    Wire.beginTransmission(QMC5883L_ADDR); Wire.write(0x0A); Wire.write(0x00); Wire.endTransmission();
    Wire.beginTransmission(QMC5883L_ADDR); Wire.write(0x0B); Wire.write(0x01); Wire.endTransmission();
}

void readQMC5883L() {
    Wire.beginTransmission(QMC5883L_ADDR); Wire.write(0x00); Wire.endTransmission(false);
    Wire.requestFrom((uint16_t)QMC5883L_ADDR, (uint8_t)6);
    if (Wire.available() == 6) {
        magX = (int16_t)(Wire.read() | (Wire.read() << 8));
        magY = (int16_t)(Wire.read() | (Wire.read() << 8));
        magZ = (int16_t)(Wire.read() | (Wire.read() << 8));
    }
}

// ---- Core 1: sensor read + threshold logic + LED/buzzer ----
void hardwareTask(void* pv) {
    initSensors();
    FastLED.addLeds<LED_CHIPSET, LED_DATA_PIN, LED_COLOR_ORD>(leds, NUM_LEDS);
    FastLED.setBrightness(LED_BRIGHT);
    float proxAvg = 0; bool proxInit = false;

    for (;;) {
        proxVal = touchRead(TOUCH_PIN);
        readQMC5883L();

        if (recalibrate) {
            baseMagX = magX; baseMagY = magY; baseMagZ = magZ;
            baseProx = proxVal; proxAvg = proxVal; proxInit = true;
            recalibrate = false;
        }
        if (proxInit) proxAvg = proxAvg * 0.999f + proxVal * 0.001f;
        baseProx = (int)proxAvg;

        int dX = magX - baseMagX, dY = magY - baseMagY, dZ = magZ - baseMagZ;
        magDelta = (int)sqrtf((float)dX * dX + (float)dY * dY + (float)dZ * dZ);

        bool proxAlert = (proxVal < threshProx);
        bool magAlert  = (magDelta > threshMag);

        if (proxAlert || magAlert) {
            const char* sev = "low";
            if (proxVal < threshProx - 1000 || magDelta > threshMag * 3) sev = "high";
            else if (proxVal < threshProx - 500 || magDelta > threshMag * 2) sev = "med";
            const char* ch = magAlert ? "mag" : "prox";
            int val = magAlert ? magDelta : proxVal;
            int thr = magAlert ? threshMag : threshProx;
            const char* kind = magAlert ? "mag_anomaly" : "proximity";
            if (!eventPending) {
                strncpy(evtChannel, ch, sizeof(evtChannel) - 1); evtChannel[sizeof(evtChannel)-1] = 0;
                strncpy(evtSeverity, sev, sizeof(evtSeverity) - 1); evtSeverity[sizeof(evtSeverity)-1] = 0;
                strncpy(evtKind, kind, sizeof(evtKind) - 1); evtKind[sizeof(evtKind)-1] = 0;
                evtValue = val; evtThreshold = thr; eventPending = true;
            }
            int pitch = constrain(map(proxVal, threshProx, 0, 400, 2000), 400, 2000);
            tone(BUZZER_PIN, pitch, 40);
            fill_solid(leds, NUM_LEDS, magAlert ? CRGB::Red : CRGB::Cyan);
        } else {
            fill_solid(leds, NUM_LEDS, CRGB::Black);
        }
        FastLED.show();
        vTaskDelay(pdMS_TO_TICKS(10));   // 100 Hz
    }
}

// ---- Core 0: network ----
WebServer        server(80);
WebSocketsServer localWS(LOCAL_WS_PORT);
WebSocketsClient appWS;
AsyncUDP         udp;

bool     isAppConnected = false, appWSStarted = false;
String   targetSSID = "", targetPASS = "", targetAppIP = "";
uint32_t bootMs = 0, seqNum = 0, lastTelemetry = 0, lastNet = 0;

void sendHello() {
    JsonDocument doc;
    doc["t"] = "hello"; doc["device"] = DEVICE_TYPE; doc["fw"] = FW_VERSION;
    doc["name"] = nodeName;
    JsonArray caps = doc["caps"].to<JsonArray>();
    caps.add("prox"); caps.add("mag"); caps.add("net");
    doc["hz"] = telemetryHz; doc["mode"] = "linked";
    JsonObject th = doc["thresholds"].to<JsonObject>();
    th["mag"] = threshMag; th["prox"] = threshProx;
    String out; serializeJson(doc, out); appWS.sendTXT(out);
}

void persistThresholds() {
    prefs.begin("specter", false);
    prefs.putInt("mag", threshMag); prefs.putInt("prox", threshProx);
    prefs.end();
}

void handleCommand(JsonDocument& doc) {
    if (doc["t"] != "cmd") return;
    const char* cmd = doc["cmd"] | "";
    if (!strcmp(cmd, "ping")) {
        JsonDocument p; p["t"] = "pong"; p["ts"] = millis();
        String out; serializeJson(p, out); if (isAppConnected) appWS.sendTXT(out);
    } else if (!strcmp(cmd, "setThreshold")) {
        const char* ch = doc["channel"] | ""; int val = doc["value"] | 0;
        if (!strcmp(ch, "mag"))  threshMag = val;
        if (!strcmp(ch, "prox")) threshProx = val;
        persistThresholds();
    } else if (!strcmp(cmd, "setRate")) {
        telemetryHz = constrain((int)(doc["hz"] | TELEMETRY_HZ_DEFAULT), 1, 30);
        prefs.begin("specter", false); prefs.putInt("hz", telemetryHz); prefs.end();
    } else if (!strcmp(cmd, "rebaseline")) {
        recalibrate = true;
    } else if (!strcmp(cmd, "identify")) {
        for (int k = 0; k < 3; k++) {
            fill_solid(leds, NUM_LEDS, CRGB::White); FastLED.show(); delay(120);
            fill_solid(leds, NUM_LEDS, CRGB::Black); FastLED.show(); delay(120);
        }
    }
}

void appWsEvent(WStype_t type, uint8_t* payload, size_t length) {
    if (type == WStype_CONNECTED) { isAppConnected = true; sendHello(); }
    else if (type == WStype_DISCONNECTED) { isAppConnected = false; }
    else if (type == WStype_TEXT) {
        JsonDocument doc; if (!deserializeJson(doc, payload, length)) handleCommand(doc);
    }
}
void sendHelloTo(uint8_t num) {
    // Same identity/thresholds the app gets, so the standalone portal can prefill.
    JsonDocument doc;
    doc["t"] = "hello"; doc["device"] = DEVICE_TYPE; doc["fw"] = FW_VERSION;
    doc["name"] = nodeName;
    JsonArray caps = doc["caps"].to<JsonArray>();
    caps.add("prox"); caps.add("mag"); caps.add("net");
    doc["hz"] = telemetryHz; doc["mode"] = "standalone";
    JsonObject th = doc["thresholds"].to<JsonObject>();
    th["mag"] = threshMag; th["prox"] = threshProx;
    String out; serializeJson(doc, out); localWS.sendTXT(num, out);
}

void localWsEvent(uint8_t num, WStype_t type, uint8_t* payload, size_t length) {
    if (type == WStype_CONNECTED) { sendHelloTo(num); }
    else if (type == WStype_TEXT) { JsonDocument doc; if (!deserializeJson(doc, payload, length)) handleCommand(doc); }
}

void connectToApp(const String& ip, int port) {
    if (appWSStarted) return;
    appWS.begin(ip, port, "/"); appWS.onEvent(appWsEvent);
    appWS.setReconnectInterval(3000); appWSStarted = true;
}

void setupNetwork() {
    prefs.begin("specter", true);
    targetSSID  = prefs.getString("ssid", "");
    targetPASS  = prefs.getString("pass", "");
    targetAppIP = prefs.getString("appip", "");
    threshMag   = prefs.getInt("mag", 1500);
    threshProx  = prefs.getInt("prox", 2000);
    telemetryHz = prefs.getInt("hz", TELEMETRY_HZ_DEFAULT);
    prefs.getString("name", "").toCharArray(nodeName, sizeof(nodeName));
    if (nodeName[0] == '\0') strncpy(nodeName, "REM-Pod A", sizeof(nodeName));
    prefs.end();

    WiFi.mode(WIFI_AP_STA);
    WiFi.softAP("SPECTER-NODE-01");
    if (targetSSID != "") {
        WiFi.begin(targetSSID.c_str(), targetPASS.c_str());
        int t = 0; while (WiFi.status() != WL_CONNECTED && t < 20) { delay(500); t++; }
    }
    if (WiFi.status() == WL_CONNECTED) {
        if (udp.listen(BEACON_PORT)) {
            udp.onPacket([](AsyncUDPPacket packet) {
                if (isAppConnected || appWSStarted) return;
                JsonDocument doc;
                if (deserializeJson(doc, packet.data(), packet.length())) return;
                if (doc["svc"] == "specter-app")
                    connectToApp(doc["ip"].as<String>(), doc["ws"] | APP_WS_PORT);
            });
        }
        if (targetAppIP != "") connectToApp(targetAppIP, APP_WS_PORT);
    }
    // ---- Standalone portal (embedded, no LittleFS) ----
    // "/"         live telemetry + 3D field-vector graph
    // "/settings" Wi-Fi, calibrate, thresholds, firmware info
    server.on("/", []()          { server.send_P(200, "text/html", PAGE_INDEX); });
    server.on("/settings", []()  { server.send_P(200, "text/html", PAGE_SETTINGS); });

    // ---- Wi-Fi management API (used by the settings page) ----
    server.on("/wifi/status", []() {
        JsonDocument d;
        bool up = (WiFi.status() == WL_CONNECTED);
        d["connected"] = up;
        d["ssid"] = up ? WiFi.SSID() : String("");
        d["rssi"] = up ? WiFi.RSSI() : 0;
        d["ip"]   = up ? WiFi.localIP().toString() : String("");
        d["ap"]   = WiFi.softAPSSID();
        String o; serializeJson(d, o); server.send(200, "application/json", o);
    });
    server.on("/wifi/scan", []() {
        int n = WiFi.scanNetworks();
        JsonDocument d; JsonArray arr = d.to<JsonArray>();
        for (int i = 0; i < n && i < 40; i++) {
            String ss = WiFi.SSID(i);
            if (ss.length() == 0) continue;
            bool seen = false;
            for (JsonObject o : arr) { if (o["ssid"] == ss) { seen = true; break; } }
            if (seen) continue;
            JsonObject o = arr.add<JsonObject>();
            o["ssid"] = ss; o["rssi"] = WiFi.RSSI(i);
            o["enc"] = (WiFi.encryptionType(i) != WIFI_AUTH_OPEN);
            if (arr.size() >= 24) break;
        }
        WiFi.scanDelete();
        String out; serializeJson(d, out); server.send(200, "application/json", out);
    });
    server.on("/wifi/save", HTTP_POST, []() {
        String ssid = server.hasArg("ssid") ? server.arg("ssid") : "";
        String pass = server.hasArg("pass") ? server.arg("pass") : "";
        ssid.trim();
        if (ssid.length() == 0) { server.send(400, "application/json", "{\"ok\":false,\"error\":\"no ssid\"}"); return; }
        WiFi.mode(WIFI_AP_STA);                 // keep AP up so the portal stays reachable
        WiFi.begin(ssid.c_str(), pass.c_str());
        int t = 0;
        while (WiFi.status() != WL_CONNECTED && t < 24) { delay(500); t++; }
        JsonDocument d;
        if (WiFi.status() == WL_CONNECTED) {
            targetSSID = ssid; targetPASS = pass;
            prefs.begin("specter", false);
            prefs.putString("ssid", ssid); prefs.putString("pass", pass);
            prefs.end();
            d["ok"] = true; d["ip"] = WiFi.localIP().toString();
        } else {
            d["ok"] = false; d["error"] = "auth or out of range";
        }
        String o; serializeJson(d, o); server.send(200, "application/json", o);
    });
    server.begin();
    localWS.begin(); localWS.onEvent(localWsEvent);
}

void setup() {
    bootMs = millis();
    xTaskCreatePinnedToCore(hardwareTask, "HW_Task", 4096, NULL, 1, NULL, 1);
    setupNetwork();
}

void loop() {
    server.handleClient(); localWS.loop(); appWS.loop();
    uint32_t now = millis();
    uint32_t period = 1000 / (telemetryHz > 0 ? telemetryHz : TELEMETRY_HZ_DEFAULT);

    if (now - lastTelemetry >= period) {
        lastTelemetry = now; seqNum++;
        JsonDocument doc;
        doc["t"] = "telemetry"; doc["ts"] = now; doc["seq"] = seqNum;
        doc["prox"] = proxVal; doc["proxBase"] = baseProx;
        JsonObject mag = doc["mag"].to<JsonObject>();
        mag["x"] = magX; mag["y"] = magY; mag["z"] = magZ;
        doc["hz"] = telemetryHz;
        String out; serializeJson(doc, out);
        if (isAppConnected) appWS.sendTXT(out);
        localWS.broadcastTXT(out);

        if (eventPending) {
            JsonDocument e;
            e["t"] = "event"; e["ts"] = now; e["kind"] = evtKind; e["channel"] = evtChannel;
            e["value"] = evtValue; e["threshold"] = evtThreshold; e["severity"] = evtSeverity;
            String eo; serializeJson(e, eo);
            if (isAppConnected) appWS.sendTXT(eo);
            localWS.broadcastTXT(eo);
            eventPending = false;
        }
    }
    if (now - lastNet >= 1000) {
        lastNet = now;
        JsonDocument n;
        n["t"] = "net"; n["ts"] = now; n["rssi"] = WiFi.RSSI();
        n["ip"] = WiFi.localIP().toString(); n["uptime"] = (now - bootMs) / 1000;
        n["ssid"] = WiFi.SSID(); n["clients"] = WiFi.softAPgetStationNum();
        String no; serializeJson(n, no);
        if (isAppConnected) appWS.sendTXT(no);
        localWS.broadcastTXT(no);
    }
}
